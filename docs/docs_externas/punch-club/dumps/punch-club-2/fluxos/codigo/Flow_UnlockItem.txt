using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Unlock Item", 0)]
[Category("Game")]
public class Flow_UnlockItem : PC2CustomFlowNode
{
	[GatherPortsCallback]
	public bool isToUnlock = true;

	[ShowIf("isToUnlock", 0)]
	public bool forever;

	[ShowIf("isToUnlock", 1)]
	public bool lockedForeverToo;

	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> itemId;

	public override string name => (isToUnlock ? "Unlock" : "Lock") + " Item";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (isToUnlock)
			{
				MainGame.me.gameSave.knowledgeSystem.UnlockItem(itemId.value, lockedForeverToo);
			}
			else
			{
				MainGame.me.gameSave.knowledgeSystem.LockItem(itemId.value, forever);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		itemId = AddValueInput<string>("itemId");
	}
}
