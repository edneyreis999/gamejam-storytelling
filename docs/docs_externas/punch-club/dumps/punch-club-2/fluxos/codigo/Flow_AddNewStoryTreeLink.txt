using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Add Story Tree Link", 0)]
[Category("Game")]
public class Flow_AddNewStoryTreeLink : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> parent;

	private ValueInput<string> child;

	protected override void RegisterPorts()
	{
		child = AddValueInput<string>("child".CapitalizeFirst());
		parent = AddValueInput<string>("parent".CapitalizeFirst());
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.knowledgeSystem.AddNewStoryLink(parent.value, child.value);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
