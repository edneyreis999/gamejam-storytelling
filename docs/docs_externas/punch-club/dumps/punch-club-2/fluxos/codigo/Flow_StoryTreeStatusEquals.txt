using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Story Tree Status Equals", 0)]
[Category("Game")]
[Color("cf35c1")]
public class Flow_StoryTreeStatusEquals : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput yes;

	private FlowOutput no;

	private ValueInput<string> storyNode;

	private ValueInput<QuestState> questState;

	private ValueOutput<bool> has;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (MainGame.me.gameSave.knowledgeSystem.GetOrCreateQuest(storyNode.value).state == questState.value)
			{
				yes.Call(flow);
			}
			else
			{
				no.Call(flow);
			}
		});
		yes = AddFlowOutput("yes".CapitalizeFirst());
		no = AddFlowOutput("no".CapitalizeFirst());
		storyNode = AddValueInput<string>("storyNode".CapitalizeFirst());
		questState = AddValueInput<QuestState>("questState".CapitalizeFirst());
		has = AddValueOutput("has", () => MainGame.me.gameSave.knowledgeSystem.GetOrCreateQuest(storyNode.value).state == questState.value);
	}
}
