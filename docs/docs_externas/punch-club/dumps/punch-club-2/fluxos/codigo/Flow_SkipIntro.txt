using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("Skip Intro", 0)]
[Category("Game")]
public class Flow_SkipIntro : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			PlayerController.Instance.ChangeCharacterActiveState(ActiveState.ByFlowScript, isEnabled: true);
			PlayerController.Instance.MovementComponent.Stop();
			ScreenCamera.me.DisableGlitch4Effect();
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
