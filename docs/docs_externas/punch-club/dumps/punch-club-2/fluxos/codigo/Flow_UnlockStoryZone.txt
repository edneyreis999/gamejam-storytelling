using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Unlock Story Zone", 0)]
[Category("Game")]
public class Flow_UnlockStoryZone : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> zoneId;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.knowledgeSystem.UnlockStoryZone(zoneId.value);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		zoneId = AddValueInput<string>("zoneId");
	}
}
