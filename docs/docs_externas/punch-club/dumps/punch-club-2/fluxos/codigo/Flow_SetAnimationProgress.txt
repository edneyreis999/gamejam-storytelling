using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Set Active Animation Progress In Seconds", 0)]
[Category("Game")]
public class Flow_SetAnimationProgress : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<Animator> animator;

	private ValueInput<string> stateName;

	private ValueInput<float> time;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			animator.value.PlayInFixedTime(stateName.value, 0, time.value);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		animator = AddValueInput<Animator>("animator");
		stateName = AddValueInput<string>("stateName");
		time = AddValueInput<float>("time");
	}
}
