using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Trigger Player Fighter Animation", 0)]
[Category("Fight")]
public class Flow_TriggerPlayerFighterAnimation : PC2CustomFlowNode
{
	private ValueInput<string> anim;

	private FlowInput @in;

	private FlowOutput @out;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			FightBehaviour.TriggerPlayerFighterAnimation(anim.value);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		anim = AddValueInput<string>("anim".CapitalizeFirst());
	}
}
