using System.Collections.Generic;
using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvasNodes;

[Name("Add Socialization To Prof Actions", 0)]
[Category("Game")]
public class Flow_AddSocializationToProfActions : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<List<string>> actions;

	private ValueInput<string> professionId;

	protected override void RegisterPorts()
	{
		professionId = AddValueInput<string>("professionId".CapitalizeFirst());
		actions = AddValueInput<List<string>>("Actions");
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (actions.value == null)
			{
				Debug.Log("Null actions list");
				@out.Call(flow);
			}
			else
			{
				float num = 0f;
				for (int i = 0; i < actions.value.Count; i++)
				{
					ProfessionActionDefinition data = GameBalance.Me.GetData<ProfessionActionDefinition>(actions.value[i]);
					ActionDefinition actionDefinition = data.ActionDefinition;
					if (data.profession != professionId.value)
					{
						Debug.Log("Action " + data.action + " not belong to profession " + professionId.value);
					}
					else
					{
						float num2 = actionDefinition.GameRes.Get(professionId.value + "_soc") * (float)data.ActionDuration;
						if (num2 < 0f)
						{
							num -= num2;
							num += 1f;
						}
					}
				}
				if (MainGame.me.PlayerRes.Get(professionId.value + "_soc") < num)
				{
					MainGame.me.PlayerRes.Set(professionId.value + "_soc", Mathf.CeilToInt(num));
				}
				@out.Call(flow);
			}
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
