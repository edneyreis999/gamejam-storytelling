using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Set Player Active", 0)]
[Category("Game/Player")]
public class Flow_SetPlayerActive : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isEnabled;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (isEnabled.value)
			{
				PlayerController.Instance.ChangeCharacterActiveState(ActiveState.ByFlowScript, isEnabled: true);
			}
			else
			{
				PlayerController.Instance.ChangeCharacterActiveState(ActiveState.ByFlowScript, isEnabled: false);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isEnabled = AddValueInput<bool>("isEnabled");
	}
}
