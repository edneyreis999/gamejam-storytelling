using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Process Achievement", 0)]
[Category("Game")]
public class Flow_ProcessAchievement : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> key;

	private ValueInput<int> counter;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			AchievementsSystem.Process(key.value, counter.value);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		key = AddValueInput<string>("key");
		counter = AddValueInput<int>("counter");
		counter.SetDefaultAndSerializedValue(1);
	}
}
