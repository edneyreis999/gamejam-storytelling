using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Send Tech Event", 0)]
[Category("Game")]
public class Flow_SendTechEvent : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> eventId;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.knowledgeSystem.ProceedEventForTechs(eventId.value);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		eventId = AddValueInput<string>("eventId");
	}
}
