using System;
using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Get Const Definition Value", 0)]
[Category("Player")]
public class Flow_GetConstDefinitionValue : PC2CustomFlowNode
{
	[GatherPortsCallback]
	public ConstsDefinition.ConstType constType;

	private ValueInput<string> definitionId;

	private ValueOutput<bool> boolValue;

	private ValueOutput<int> intValue;

	private ValueOutput<float> floatValue;

	private ValueOutput<string> stringValue;

	protected override void RegisterPorts()
	{
		definitionId = AddValueInput<string>("definitionId".CapitalizeFirst());
		switch (constType)
		{
		case ConstsDefinition.ConstType.@bool:
			boolValue = AddValueOutput("boolValue".CapitalizeFirst(), () => ConstsDefinition.Get(definitionId.value).BoolValue);
			break;
		case ConstsDefinition.ConstType.@int:
			intValue = AddValueOutput("intValue".CapitalizeFirst(), () => ConstsDefinition.Get(definitionId.value).IntValue);
			break;
		case ConstsDefinition.ConstType.@float:
			floatValue = AddValueOutput("floatValue".CapitalizeFirst(), () => ConstsDefinition.Get(definitionId.value).FloatValue);
			break;
		case ConstsDefinition.ConstType.@string:
			stringValue = AddValueOutput("stringValue".CapitalizeFirst(), () => ConstsDefinition.Get(definitionId.value).StringValue);
			break;
		default:
			throw new ArgumentOutOfRangeException();
		}
	}
}
