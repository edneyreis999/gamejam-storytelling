using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("UI League Show Empty Mask", 0)]
[Category("Game/Tutorial")]
public class Flow_UILeagueShowEmptyMask : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isToShow;

	public override string name => "UI League " + (isToShow.value ? "Show" : "Hide") + " Empty Mask";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			RectTransform emptyMask = GUIElements.Get<UILeagueWindow>().EmptyMask;
			if (isToShow.value)
			{
				GUIElements.Get<UITutorialWindow>().ShowMask(emptyMask);
			}
			else
			{
				GUIElements.Get<UITutorialWindow>().HideMask(emptyMask);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isToShow = AddValueInput<bool>("isToShow");
	}
}
