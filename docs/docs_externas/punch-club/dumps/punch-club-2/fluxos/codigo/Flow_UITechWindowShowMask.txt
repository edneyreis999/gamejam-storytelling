using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("UI Tech Window Show Mask", 0)]
[Category("Game/Tutorial")]
public class Flow_UITechWindowShowMask : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isToShow;

	public override string name => "UI Tech Window " + (isToShow.value ? "Show" : "Hide") + " Mask";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			RectTransform techScrollViewRectTutorial = GUIElements.Get<UITechWindow>().TechScrollViewRectTutorial;
			if (isToShow.value)
			{
				GUIElements.Get<UITutorialWindow>().ShowMask(techScrollViewRectTutorial);
			}
			else
			{
				GUIElements.Get<UITutorialWindow>().HideMask(techScrollViewRectTutorial);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isToShow = AddValueInput<bool>("isToShow");
	}
}
