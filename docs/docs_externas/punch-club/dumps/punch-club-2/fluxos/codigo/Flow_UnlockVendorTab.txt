using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Unlock Vendor Tab", 0)]
[Category("Game")]
public class Flow_UnlockVendorTab : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> tabId;

	private ValueInput<int> sortingPriority;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.uiSystem.UnlockVendorTab(new VendorTabData
			{
				tabId = tabId.value,
				sortingOrder = sortingPriority.value
			});
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		tabId = AddValueInput<string>("tabId");
		sortingPriority = AddValueInput<int>("sortingPriority");
	}
}
