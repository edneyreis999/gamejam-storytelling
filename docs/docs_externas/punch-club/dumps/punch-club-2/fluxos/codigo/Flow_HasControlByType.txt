using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Has Control By Type", 0)]
[Category("Game")]
[Color("cf35c1")]
public class Flow_HasControlByType : PC2CustomFlowNode
{
	[GatherPortsCallback]
	public ControlState controlState;

	private FlowInput @in;

	private FlowOutput yes;

	private FlowOutput no;

	public override string name => $"Has Control {controlState}?";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (PlayerController.Instance.HasControl(controlState))
			{
				yes.Call(flow);
			}
			else
			{
				no.Call(flow);
			}
		});
		yes = AddFlowOutput("yes".CapitalizeFirst());
		no = AddFlowOutput("no".CapitalizeFirst());
	}
}
