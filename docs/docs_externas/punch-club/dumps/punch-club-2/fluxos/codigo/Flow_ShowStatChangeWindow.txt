using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Show Stat Change Window", 0)]
[Category("Game/UI")]
public class Flow_ShowStatChangeWindow : PC2CustomFlowNode
{
	[GatherPortsCallback]
	public bool isToHide;

	private FlowInput @in;

	private FlowOutput @out;

	public override string name
	{
		get
		{
			if (!isToHide)
			{
				return "Show Stat Change Window";
			}
			return "Hide Stat Change Window";
		}
	}

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			GUIElements.Get<UIStatChangeWindow>().Open();
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
