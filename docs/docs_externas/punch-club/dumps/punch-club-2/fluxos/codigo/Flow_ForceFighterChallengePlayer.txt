using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Force Fighter Challenge Player", 0)]
[Category("Game/Dialogue")]
public class Flow_ForceFighterChallengePlayer : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> fighterId;

	private ValueInput<string> leagueId;

	private ValueInput<bool> forceValue;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in", delegate(Flow flow)
		{
			League league = MainGame.me.gameSave.leaguesSystem.GetLeague(leagueId.value);
			if (league == null)
			{
				Debug.Log("#league# Can't force fighter:[" + fighterId.value + "] from null league:[" + leagueId.value + "]");
			}
			else
			{
				LeagueFighter fighterById = league.GetFighterById(fighterId.value);
				if (fighterById == null)
				{
					Debug.Log("#league# Can't force null fighter:[" + fighterId.value + "] from league:[" + leagueId.value + "]");
				}
				else
				{
					fighterById.forceChallengePlayer = forceValue.value;
				}
			}
			@out.Call(flow);
		});
		fighterId = AddValueInput<string>("fighterId".CapitalizeFirst());
		leagueId = AddValueInput<string>("leagueId".CapitalizeFirst());
		forceValue = AddValueInput<bool>("forceValue".CapitalizeFirst());
		@out = AddFlowOutput("out");
	}
}
