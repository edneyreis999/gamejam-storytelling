using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Change League's Widget State\nIn Character Window", 0)]
[Category("Game/UI")]
public class Flow_UnlockLeaguesInCharacterWindow : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isActive;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.uiSystem.uiCharacterWindowData.isLeaguesLocked = !isActive.value;
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isActive = AddValueInput<bool>("isActive");
	}
}
