using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("Set Robot State", 0)]
public class Flow_SetRobotState : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<RobotState> robotState;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.robotSystem.CurrentState = robotState.value;
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		robotState = AddValueInput<RobotState>("robotState");
	}
}
