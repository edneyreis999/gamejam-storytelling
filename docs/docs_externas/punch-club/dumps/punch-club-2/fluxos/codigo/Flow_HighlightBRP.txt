using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("Show Mask BRP", 0)]
[Category("Game/Tutorial")]
public class Flow_HighlightBRP : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isToShow;

	public override string name => "BRP" + (isToShow.value ? "Show" : "Hide") + " Mask";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			RectTransform brpRectTransform = GUIElements.Get<HUD>().RobotHUDWidget.brpRectTransform;
			if (isToShow.value)
			{
				GUIElements.Get<UITutorialWindow>().ShowMask(brpRectTransform);
			}
			else
			{
				GUIElements.Get<UITutorialWindow>().HideMask(brpRectTransform);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isToShow = AddValueInput<bool>("isToShow");
	}
}
