using System.Collections.Generic;
using FlowCanvas;
using LazyBearTechnology;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvasNodes;

[Name("Add Required Stats To Prof Actions", 0)]
[Category("Game")]
public class Flow_AddAllRequiredStatsToProfActions : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<List<string>> actions;

	protected override void RegisterPorts()
	{
		actions = AddValueInput<List<string>>("Actions");
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (actions.value == null)
			{
				Debug.Log("Null actions list");
				@out.Call(flow);
			}
			else
			{
				float num = 0f;
				float num2 = 0f;
				float num3 = 0f;
				float num4 = 0f;
				for (int i = 0; i < actions.value.Count; i++)
				{
					ProfessionActionDefinition data = GameBalance.Me.GetData<ProfessionActionDefinition>(actions.value[i]);
					GameRes gameRes = data.ActionDefinition.GameRes;
					float num5 = gameRes.Get("_satiety") * (float)data.ActionDuration;
					float num6 = gameRes.Get("_health") * (float)data.ActionDuration;
					float num7 = gameRes.Get("_satisfaction") * (float)data.ActionDuration;
					float num8 = gameRes.Get("_energy") * (float)data.ActionDuration;
					if (num5 < 0f)
					{
						num -= num5;
						num += 1f;
					}
					if (num6 < 0f)
					{
						num2 -= num6;
						num2 += 1f;
					}
					if (num7 < 0f)
					{
						num4 -= num7;
						num4 += 1f;
					}
					if (num8 < 0f)
					{
						num3 -= num8;
						num3 += 1f;
					}
				}
				float num9 = MainGame.me.PlayerRes.Get("_satiety");
				float num10 = MainGame.me.PlayerRes.Get("_health");
				float num11 = MainGame.me.PlayerRes.Get("_satisfaction");
				float num12 = MainGame.me.PlayerRes.Get("_energy");
				if (num9 < num)
				{
					MainGame.me.PlayerRes.Set("_satiety", Mathf.CeilToInt(num));
				}
				if (num10 < num2)
				{
					MainGame.me.PlayerRes.Set("_health", Mathf.CeilToInt(num2));
				}
				if (num11 < num4)
				{
					MainGame.me.PlayerRes.Set("_satisfaction", Mathf.CeilToInt(num4));
				}
				if (num12 < num3)
				{
					MainGame.me.PlayerRes.Set("_energy", Mathf.CeilToInt(num3));
				}
				@out.Call(flow);
			}
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
