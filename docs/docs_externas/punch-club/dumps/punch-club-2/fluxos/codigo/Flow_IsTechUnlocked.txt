using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("UI Is Tech Unlocked", 0)]
[Category("Game")]
public class Flow_IsTechUnlocked : PC2CustomFlowNode
{
	private ValueInput<string> tech;

	private FlowInput @in;

	private FlowOutput yes;

	private FlowOutput no;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (MainGame.me.gameSave.knowledgeSystem.IsTechObtained(tech.value))
			{
				yes.Call(flow);
			}
			else
			{
				no.Call(flow);
			}
		});
		yes = AddFlowOutput("yes".CapitalizeFirst());
		no = AddFlowOutput("no".CapitalizeFirst());
		tech = AddValueInput<string>("tech".CapitalizeFirst());
	}
}
