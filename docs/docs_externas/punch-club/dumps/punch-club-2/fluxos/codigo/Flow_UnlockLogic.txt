using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Unlock Logic", 0)]
[Category("Game")]
public class Flow_UnlockLogic : PC2CustomFlowNode
{
	[GatherPortsCallback]
	public bool isToUnlock = true;

	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> logicId;

	public override string name => (isToUnlock ? "Unlock" : "Lock") + " Logic";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (isToUnlock)
			{
				GameLogics.UnlockLogic(logicId.value);
			}
			else
			{
				GameLogics.LockLogic(logicId.value);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		logicId = AddValueInput<string>("logicId");
	}
}
