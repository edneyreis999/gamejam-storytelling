using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Try To Move Item", 0)]
[Category("Game")]
public class Flow_TryToMoveItem : PC2CustomFlowNode
{
	[GatherPortsCallback]
	public bool useAllInventories = true;

	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> inventoryID;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), TryToMoveItem);
		@out = AddFlowOutput("out".CapitalizeFirst());
		if (!useAllInventories)
		{
			inventoryID = AddValueInput<string>("inventoryID".CapitalizeFirst());
		}
	}

	private void TryToMoveItem(Flow flow)
	{
		if (useAllInventories)
		{
			for (int i = 0; i < GameBalance.Me.inventoriesData.Count; i++)
			{
				InventoryUtils.FindRelocatableItemsAndTryToMoveThem(GameBalance.Me.inventoriesData[i].id);
			}
		}
		else
		{
			InventoryUtils.FindRelocatableItemsAndTryToMoveThem(inventoryID.value);
		}
		@out.Call(flow);
	}
}
