using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("UI Show Duty Progress Mask", 0)]
[Category("Game/Tutorial")]
public class Flow_UIProfWidgetShowDutySelection : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isToShow;

	public override string name => "UI " + (isToShow.value ? "Show" : "Hide") + " Duty Progress Mask";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			RectTransform dutyProgressTutorialRect = GUIElements.Get<HUD>().ProfessionProgressWidget.DutyProgressTutorialRect;
			if (isToShow.value)
			{
				GUIElements.Get<UITutorialWindow>().ShowMask(dutyProgressTutorialRect);
			}
			else
			{
				GUIElements.Get<UITutorialWindow>().HideMask(dutyProgressTutorialRect);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isToShow = AddValueInput<bool>("isToShow");
	}
}
