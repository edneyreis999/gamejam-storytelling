using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Open RepValue Window", 0)]
[Category("Game")]
public class Flow_OpenRepValueWindow : PC2CustomFlowNode
{
	[GatherPortsCallback]
	public bool isToHide;

	private FlowInput @in;

	private FlowOutput @out;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in", delegate(Flow flow)
		{
			UIRepValue uIRepValue = GUIElements.Get<UIRepValue>();
			if (isToHide)
			{
				uIRepValue.Hide();
			}
			else
			{
				uIRepValue.Open();
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
