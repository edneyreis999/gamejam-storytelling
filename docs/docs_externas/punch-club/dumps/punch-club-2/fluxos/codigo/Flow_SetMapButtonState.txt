using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Set Map Button State", 0)]
[Category("Game")]
public class Flow_SetMapButtonState : PC2CustomFlowNode
{
	public bool interactable;

	public bool changeVisibility;

	[ShowIf("changeVisibility", 1)]
	public bool isVisible;

	private FlowInput @in;

	private FlowOutput @out;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.uiSystem.ChangeMapButtonInteractableState(interactable);
			if (changeVisibility)
			{
				MainGame.me.gameSave.uiSystem.ChangeMapButtonState(isVisible);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
