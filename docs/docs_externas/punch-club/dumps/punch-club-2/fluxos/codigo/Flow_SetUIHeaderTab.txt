using System;
using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Set UIHeaderWindow Tab", 0)]
[Category("Game")]
public class Flow_SetUIHeaderTab : PC2CustomFlowNode
{
	public HeaderTabType headerTabType;

	private FlowInput @in;

	private FlowOutput @out;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			UIHeaderWindow uIHeaderWindow = GUIElements.Get<UIHeaderWindow>();
			uIHeaderWindow.CurrentTab = headerTabType;
			switch (headerTabType)
			{
			case HeaderTabType.Character:
				uIHeaderWindow.CurrentOpenedTabWindow = GUIElements.Get<UICharacterWindow>();
				break;
			case HeaderTabType.Fighter:
				uIHeaderWindow.CurrentOpenedTabWindow = GUIElements.Get<UIFighterWindow>();
				break;
			case HeaderTabType.Tech:
				uIHeaderWindow.CurrentOpenedTabWindow = GUIElements.Get<UITechWindow>();
				break;
			case HeaderTabType.Story:
				uIHeaderWindow.CurrentOpenedTabWindow = GUIElements.Get<UIStoryTreeWindow>();
				break;
			case HeaderTabType.Software:
				uIHeaderWindow.CurrentOpenedTabWindow = GUIElements.Get<UISoftwareWindow>();
				break;
			case HeaderTabType.Robot:
				uIHeaderWindow.CurrentOpenedTabWindow = GUIElements.Get<RobotWindow>();
				break;
			default:
				throw new ArgumentOutOfRangeException();
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
