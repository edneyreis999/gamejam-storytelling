using FlowCanvas;
using LazyBearTechnology;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Open Robot Training Window", 0)]
[Category("Game/UI")]
public class Flow_RobotTrainingWindowOpen : PC2CustomFlowNode
{
	[GatherPortsCallback]
	public bool isToOpen;

	private FlowInput @in;

	private FlowOutput @out;

	private FlowOutput onWindowOpened;

	private FlowOutput onWindowClosed;

	private FlowOutput onNotSelectedClosed;

	private FlowOutput onSelected;

	private ValueInput<string> vendorId;

	private ValueInput<WGO> paramWgo;

	private Flow cachedFlow;

	public override string name => (isToOpen ? "Open " : "Close ") + "Robot Training Window";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), ToggleWindow);
		@out = AddFlowOutput("out".CapitalizeFirst());
		if (isToOpen)
		{
			onWindowOpened = AddFlowOutput("onWindowOpened".CapitalizeFirst());
			onWindowClosed = AddFlowOutput("onWindowClosed".CapitalizeFirst());
			onNotSelectedClosed = AddFlowOutput("onNotSelectedClosed".CapitalizeFirst());
			onSelected = AddFlowOutput("onSelected".CapitalizeFirst());
			vendorId = AddValueInput<string>("vendorId".CapitalizeFirst());
			paramWgo = AddValueInput<WGO>("paramWgo".CapitalizeFirst());
		}
	}

	private void ToggleWindow(Flow flow)
	{
		cachedFlow = flow;
		UIRobotTrainingWindow robotTrainingWindow;
		if (isToOpen)
		{
			robotTrainingWindow = GUIElements.Get<UIRobotTrainingWindow>();
			BaseGUI.OnWindowOpened += OnWindowOpened;
			BaseGUI.OnWindowClosed += OnWindowClosed;
			UIRobotTrainingWindow.onNotAnyTrainingSelected += OnNotSelected;
			UIRobotTrainingWindow.onAnyTrainingSelected += OnSelected;
			robotTrainingWindow.Open(vendorId.value, WGOParamOrSelf(paramWgo));
		}
		else
		{
			GUIElements.Get<UIRobotTrainingWindow>().Hide();
		}
		@out.Call(flow);
		void OnWindowClosed(BaseGUI window)
		{
			if (window is UIRobotTrainingWindow)
			{
				onWindowClosed.Call(flow);
				BaseGUI.OnWindowClosed -= OnWindowClosed;
			}
		}
		void OnWindowOpened(BaseGUI window)
		{
			if (window is UIRobotTrainingWindow)
			{
				LazyTimer.AddTimer(robotTrainingWindow.WindowAnimationTime, delegate
				{
					onWindowOpened.Call(flow);
					BaseGUI.OnWindowOpened -= OnWindowOpened;
				});
			}
		}
	}

	private void OnNotSelected()
	{
		onNotSelectedClosed.Call(cachedFlow);
		UIRobotTrainingWindow.onNotAnyTrainingSelected -= OnNotSelected;
	}

	private void OnSelected()
	{
		onSelected.Call(cachedFlow);
		UIRobotTrainingWindow.onAnyTrainingSelected -= OnSelected;
	}
}
