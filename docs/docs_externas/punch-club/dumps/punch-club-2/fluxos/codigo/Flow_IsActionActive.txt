using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Is Action Active", 0)]
[Category("Game")]
[Color("cf35c1")]
public class Flow_IsActionActive : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput yes;

	private FlowOutput no;

	private ValueInput<string> actionId;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (MainGame.me.ActionProcessor.IsActive && MainGame.me.ActionProcessor.ActiveAction.id == actionId.value)
			{
				yes.Call(flow);
			}
			else
			{
				no.Call(flow);
			}
		});
		actionId = AddValueInput<string>("actionId".CapitalizeFirst());
		yes = AddFlowOutput("yes".CapitalizeFirst());
		no = AddFlowOutput("no".CapitalizeFirst());
	}
}
