using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Is Tape Watched", 0)]
[Category("Game")]
[Color("cf35c1")]
public class Flow_IsTapeWatched : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput yes;

	private FlowOutput no;

	private ValueInput<string> tapeId;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (MainGame.me.gameSave.playerData.IsTapeWatched(tapeId.value))
			{
				yes.Call(flow);
			}
			else
			{
				no.Call(flow);
			}
		});
		tapeId = AddValueInput<string>("tapeId".CapitalizeFirst());
		yes = AddFlowOutput("yes".CapitalizeFirst());
		no = AddFlowOutput("no".CapitalizeFirst());
	}
}
