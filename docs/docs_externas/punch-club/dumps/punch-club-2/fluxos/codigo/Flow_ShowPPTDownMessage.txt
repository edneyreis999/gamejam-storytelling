using FlowCanvas;
using LazyBearTechnology;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("Show PPT Down Message", 0)]
[Category("Game")]
public class Flow_ShowPPTDownMessage : PC2CustomFlowNode
{
	[GatherPortsCallback]
	public bool isToHide;

	private FlowInput @in;

	private FlowOutput @out;

	private FlowOutput onFinished;

	private FlowOutput onTextAnimationEnd;

	private ValueInput<string> text;

	private ValueInput<string> speaker;

	private ValueInput<int> voiceId;

	public override string name
	{
		get
		{
			if (!isToHide)
			{
				return "Show PPT Down Message";
			}
			return "Hide PPT Down Message";
		}
	}

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			UIPPTWindow uIPPTWindow = GUIElements.Get<UIPPTWindow>();
			if (isToHide)
			{
				uIPPTWindow.HideDownMessage(delegate
				{
					onFinished.Call(flow);
				});
			}
			else
			{
				VoiceID voiceID = new VoiceID(voiceId.value);
				uIPPTWindow.ShowDownMessage(speaker.value, text.value, voiceID, delegate
				{
					onTextAnimationEnd.Call(flow);
				}, delegate
				{
					onFinished.Call(flow);
				});
			}
			@out.Call(flow);
		});
		if (!isToHide)
		{
			voiceId = AddValueInput<int>("voiceId".CapitalizeFirst());
			voiceId.SetDefaultAndSerializedValue(1);
			text = AddValueInput<string>("text".CapitalizeFirst());
			speaker = AddValueInput<string>("speaker".CapitalizeFirst());
			onTextAnimationEnd = AddFlowOutput("onTextAnimationEnd".CapitalizeFirst());
		}
		onFinished = AddFlowOutput("onFinished".CapitalizeFirst());
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
