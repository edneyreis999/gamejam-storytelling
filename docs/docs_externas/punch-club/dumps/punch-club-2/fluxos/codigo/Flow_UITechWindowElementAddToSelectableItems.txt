using System.Collections.Generic;
using FlowCanvas;
using LazyBearTechnology;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("UI Tech Window Element Add to Selectable Items", 0)]
[Category("Game/Tutorial")]
public class Flow_UITechWindowElementAddToSelectableItems : PC2CustomFlowNode
{
	public List<string> elementsId;

	private FlowInput @in;

	private FlowOutput @out;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			List<GamepadNavigationItem> list = new List<GamepadNavigationItem>();
			UITechWindow uITechWindow = GUIElements.Get<UITechWindow>();
			foreach (string item in elementsId)
			{
				UITechElement elementById = uITechWindow.GetElementById(item);
				list.Add(elementById.Item);
			}
			uITechWindow.CreateCustomSelectableItems(list);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
