using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Enable All Available Scholars", 0)]
[Category("Game/MickGym")]
public class Flow_EnableAllAvailableScholars : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	public bool notEnableGameObjects;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			int num = 0;
			for (int i = 0; i < LazyConsts.mickScholarsIds.Count; i++)
			{
				WGOState wGOState = MainGame.me.gameSave.worldData.GetWGOState(LazyConsts.mickScholarsTrainersIds[i]);
				if (wGOState != null && wGOState.GetStateOnScene("MikesOldHouse", out var isActive))
				{
					WGOState wGOState2 = MainGame.me.gameSave.worldData.GetWGOState(LazyConsts.mickScholarsIds[i]);
					if (wGOState2 != null && isActive && wGOState2.gameRes.GetInt("is_unlocked") == 1)
					{
						wGOState2.SetStateOnScene("MikesOldHouse", isActive: true);
						WGO wGO = DynamicScene.FindWGO(LazyConsts.mickScholarsIds[i]);
						if (wGO != null && !notEnableGameObjects)
						{
							wGO.gameObject.SetActive(value: true);
						}
						num++;
					}
				}
			}
			MainGame.me.PlayerRes.Set("scholars_count", num);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
