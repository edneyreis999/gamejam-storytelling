using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Despawn WGO", 0)]
[Category("Game/WGO")]
public class Flow_DespawnWGO : PC2CustomFlowNode
{
	public enum DespawnBy
	{
		Id,
		Ref
	}

	[GatherPortsCallback]
	public DespawnBy despawnByType = DespawnBy.Ref;

	private FlowInput @in;

	private FlowOutput @out;

	private new ValueInput<WGO> wgo;

	private ValueInput<string> wgoId;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), Despawn);
		@out = AddFlowOutput("out");
		switch (despawnByType)
		{
		case DespawnBy.Id:
			wgoId = AddValueInput<string>("wgoId");
			break;
		case DespawnBy.Ref:
			wgo = AddValueInput<WGO>("WGO");
			break;
		}
	}

	private void Despawn(Flow flow)
	{
		switch (despawnByType)
		{
		case DespawnBy.Id:
			DynamicScene.DespawnWGO(wgoId.value);
			break;
		case DespawnBy.Ref:
		{
			WGO wGO = WGOParamOrSelf(wgo);
			if (wGO != null)
			{
				DynamicScene.DespawnWGO(wGO);
			}
			else
			{
				Debug.LogError("Tried to despawn null WGO");
			}
			break;
		}
		}
		@out.Call(flow);
	}
}
