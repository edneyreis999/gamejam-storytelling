using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Remove Profession", 0)]
[Category("Game")]
public class Flow_RemoveProfession : PC2CustomFlowNode
{
	public bool toPool;

	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> professionId;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.professionSystem.RemoveProfession(professionId.value, toPool);
			DynamicScene.current.UpdateProfessionWidgetState();
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		professionId = AddValueInput<string>("professionId".CapitalizeFirst());
	}
}
