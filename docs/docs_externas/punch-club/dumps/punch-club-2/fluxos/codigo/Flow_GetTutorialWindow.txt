using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvasNodes;

[Name("Get Tutorial Window", 0)]
[Category("Game")]
public class Flow_GetTutorialWindow : PC2CustomFlowNode
{
	private ValueOutput<GameObject> tutorialWindowObject;

	protected override void RegisterPorts()
	{
		tutorialWindowObject = AddValueOutput("tutorialWindowObject".CapitalizeFirst(), () => GUIElements.Get<UITutorialWindow>().gameObject);
	}
}
