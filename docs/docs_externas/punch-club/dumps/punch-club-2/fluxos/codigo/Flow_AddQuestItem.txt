using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Add Quest Item", 0)]
[Category("Game")]
public class Flow_AddQuestItem : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> itemId;

	private ValueInput<bool> notShowNotice;

	[GatherPortsCallback]
	public bool isToRemove;

	public override string name => (isToRemove ? "Remove" : "Add") + " Quest Item";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (isToRemove)
			{
				InventoryUtils.RemoveItemFromInventory(itemId.value, 1, !notShowNotice.value);
			}
			else
			{
				InventoryUtils.AddItemToInventory(itemId.value, 1, !notShowNotice.value);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		itemId = AddValueInput<string>("itemId");
		notShowNotice = AddValueInput<bool>("notShowNotice");
	}
}
