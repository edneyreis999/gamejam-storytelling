using System.Collections.Generic;
using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("UI Set Tech Elements Interactable", 0)]
[Category("Game/Tutorial")]
public class Flow_SetTechElementInteractable : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isInteractable;

	private ValueInput<List<string>> ids;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			UITechWindow uITechWindow = GUIElements.Get<UITechWindow>();
			foreach (string item in ids.value)
			{
				uITechWindow.GetElementById(item).IsNotInteractable = !isInteractable.value;
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		ids = AddValueInput<List<string>>("ids");
		isInteractable = AddValueInput<bool>("isInteractable");
	}
}
