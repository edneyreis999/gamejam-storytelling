using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Add Perk", 0)]
[Category("Game")]
public class Flow_AddPerk : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<string> perkId;

	private ValueInput<bool> showNotice;

	protected override void RegisterPorts()
	{
		perkId = AddValueInput<string>("perkId".CapitalizeFirst());
		showNotice = AddValueInput<bool>("showNotice".CapitalizeFirst());
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.playerData.AddPerk(perkId.value, showNotice.value);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
