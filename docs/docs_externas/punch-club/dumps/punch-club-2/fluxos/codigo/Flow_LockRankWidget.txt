using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Lock Rank Widget", 0)]
[Category("Game")]
public class Flow_LockRankWidget : PC2CustomFlowNode
{
	[GatherPortsCallback]
	public bool isToUnlock = true;

	private FlowInput @in;

	private FlowOutput @out;

	public override string name => (isToUnlock ? "Unlock" : "Lock") + " Rank Widget";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (isToUnlock)
			{
				MainGame.me.gameSave.uiSystem.isPlayerRankWidgetEnabled = true;
			}
			else
			{
				MainGame.me.gameSave.uiSystem.isPlayerRankWidgetEnabled = false;
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
