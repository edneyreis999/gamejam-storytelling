using FlowCanvas;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Hide MultiAnswer", 0)]
[Category("Game/Dialogue")]
[Icon("DialogueSmall", false, "")]
[Color("00EE00")]
public class Flow_HideMultianswer : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in", delegate(Flow flow)
		{
			Bubble.HideMultiAnswer();
			@out.Call(flow);
		});
		@out = AddFlowOutput("out");
	}
}
