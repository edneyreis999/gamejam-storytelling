	public static void LoadGameBalance()
	{
		instance = Resources.Load<GameBalance>("GameBalance");
		if (instance == null)
		{
			Debug.LogError("Game data load failed");
			return;
		}
		instance.CreateIDsCache();
		instance.stmLevel.levels.Clear();
		instance.stmLevel.id = "stm";
		instance.aglLevel.levels.Clear();
		instance.aglLevel.id = "agl";
		instance.strLevel.levels.Clear();
		instance.strLevel.id = "str";
		foreach (LevelCurvesDefinition levelCurvesDatum in instance.levelCurvesData)
		{
			instance.aglLevel.levels.Add(levelCurvesDatum.agl);
			instance.aglLevel.fuels.Add(levelCurvesDatum.aglFuel);
			instance.strLevel.levels.Add(levelCurvesDatum.str);
			instance.strLevel.fuels.Add(levelCurvesDatum.strFuel);
			instance.stmLevel.levels.Add(levelCurvesDatum.stm);
			instance.stmLevel.fuels.Add(levelCurvesDatum.stmFuel);
		}
	}

	public ActionDefinition GetActionDefinitionByLinkedWGO(string wgoId)
	{
		foreach (ActionDefinition actionDatum in actionData)
		{
			if (actionDatum.linkedWgo == wgoId)
			{
				return actionDatum;
			}
		}
		Debug.LogError("Cannot find ActionDefinition for linked wgo id: " + wgoId);
		return null;
	}

	public OldSchoolTrainingDefinition GetOldSchoolTrainingDefinitionByLinkedWGO(string wgoId)
	{
		foreach (OldSchoolTrainingDefinition oldSchoolTrainingDatum in oldSchoolTrainingData)
		{
			if (oldSchoolTrainingDatum.linkedWgo == wgoId)
			{
				return oldSchoolTrainingDatum;
			}
		}
		Debug.LogError("Cannot find OldSchoolTrainingDefinition for linked wgo id: " + wgoId);
		return null;
	}

	public List<ItemDefinition> GetItemsByTypes(List<string> types)
	{
		List<ItemDefinition> list = new List<ItemDefinition>();
		for (int i = 0; i < itemsData.Count; i++)
		{
			if (itemsData[i].HasAnyType(types))
			{
				list.Add(itemsData[i]);
			}
		}
		return list;
	}

	public FightSchoolDefinition GetSchoolById(string schoolId)
	{
		for (int i = 0; i < fightSchoolData.Count; i++)
		{
			FightSchoolDefinition fightSchoolDefinition = fightSchoolData[i];
			if (schoolId == fightSchoolDefinition.id)
			{
				return fightSchoolDefinition;
			}
		}
		return null;
	}

	public PerkDefinition GetPerkById(string perkId)
	{
		if (string.IsNullOrEmpty(perkId))
		{
			return null;
		}
		if (perkId.StartsWith("gold_buff_"))
		{
			PerkDefinition perkDefinition = cachedGoldBuffs.Find((PerkDefinition p) => p.id == perkId);
			if (perkDefinition != null)
			{
				return perkDefinition;
			}
			perkDefinition = new PerkDefinition();
			perkDefinition.id = perkId;
			perkDefinition.description = perkId + "_d";
			perkDefinition.definiotionType = PerkDefinitionType.Fight;
			perkDefinition.duration = 4320;
			perkDefinition.sortingOrder = 0;
			string text = perkId.Replace("d_act_", "");
			text = text.Replace("act_", "");
			perkDefinition.icon = text;
			perkDefinition.type = PerkType.AsNew;
			cachedGoldBuffs.Add(perkDefinition);
			return perkDefinition;
		}
		for (int num = 0; num < perkData.Count; num++)
		{
			PerkDefinition perkDefinition = perkData[num];
			if (perkId == perkDefinition.id)
			{
				return perkDefinition;
			}
		}
		return null;
	}

	public FightActionDataDefinition GetFightActionById(string actionId)
	{
		for (int i = 0; i < fightActionsDataList.Count; i++)
		{
			FightActionDataDefinition fightActionDataDefinition = fightActionsDataList[i];
			if (actionId == fightActionDataDefinition.id)
			{
				return fightActionDataDefinition;
			}
		}
		return null;
	}

	public int GetVoiceIdByWGOId(string wgoId)
	{
		for (int i = 0; i < wgoData.Count; i++)
		{
			if (wgoData[i].id == wgoId)
			{
				return wgoData[i].voiceId;
			}
		}
		Debug.LogError("Can't find VoiceId for WGO with id: " + wgoId);
		return 0;
	}
