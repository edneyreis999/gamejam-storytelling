using FlowCanvas;
using LazyBearTechnology;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvasNodes;

[Name("Enable LazyAnimationController", 0)]
[Category("Game")]
public class Flow_EnableLazyAnimationController : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<GameObject> gameObject;

	private ValueInput<bool> isEnable;

	public override string name => (isEnable.value ? "Enable" : "Disable") + " LazyAnimationController";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), Toggle);
		@out = AddFlowOutput("out".CapitalizeFirst());
		gameObject = AddValueInput<GameObject>("gameObject".CapitalizeFirst());
		isEnable = AddValueInput<bool>("isEnable".CapitalizeFirst());
	}

	private void Toggle(Flow flow)
	{
		GameObject gameObject = null;
		gameObject = (this.gameObject.isConnected ? this.gameObject.value : CustomFlowNode.GetGameObjectFromNode(this));
		if (gameObject != null)
		{
			LazyAnimationController componentInChildren = gameObject.gameObject.GetComponentInChildren<LazyAnimationController>(includeInactive: true);
			if (componentInChildren != null)
			{
				componentInChildren.enabled = isEnable.value;
			}
			else
			{
				Debug.LogError("Tried to toggle LazyAnimationController: not found component");
			}
		}
		else
		{
			Debug.LogError("Tried to toggle LazyAnimationController: null object");
		}
		@out.Call(flow);
	}
}
