using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("Set HUD Interactable", 0)]
[Category("Game/Tutorial")]
public class Flow_SetHUDInteractable : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isInteractable;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			GUIElements.Get<HUD>().Interactable = isInteractable.value;
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isInteractable = AddValueInput<bool>("isInteractable".CapitalizeFirst());
	}
}
