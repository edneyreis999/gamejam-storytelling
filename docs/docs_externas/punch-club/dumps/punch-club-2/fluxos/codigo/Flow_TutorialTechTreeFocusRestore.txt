using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Try Restore Tech Tree Gamepad Focus", 0)]
[Category("Game")]
public class Flow_TutorialTechTreeFocusRestore : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	public bool reinit;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			UITechWindow uITechWindow = GUIElements.Get<UITechWindow>();
			if (reinit)
			{
				uITechWindow.UpdateGamepadFocus();
			}
			else
			{
				uITechWindow.TryRestoreGamepadFocus();
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
