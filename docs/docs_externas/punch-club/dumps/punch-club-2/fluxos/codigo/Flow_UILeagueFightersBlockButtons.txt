using System.Collections.Generic;
using FlowCanvas;
using LazyBearTechnology;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("UI League Fighters\nBlock Buttons", 0)]
[Category("Game/Tutorial")]
public class Flow_UILeagueFightersBlockButtons : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isToBlock;

	public override string name => "UI League Fighters\n" + (isToBlock.value ? "Block" : "Unblock") + " Buttons";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			UILeagueWindow uILeagueWindow = GUIElements.Get<UILeagueWindow>();
			uILeagueWindow.EloLeague.SetFighterButtonsInteractionState(!isToBlock.value);
			uILeagueWindow.CustomItems = new List<GamepadNavigationItem> { uILeagueWindow.emptyTutorialItem };
			uILeagueWindow.TutorialGamepadNavigation = true;
			uILeagueWindow.LockPressBack = true;
			uILeagueWindow.LockCompare = true;
			uILeagueWindow.CustomLockPressBack = true;
			uILeagueWindow.LockSwitchTab = true;
			uILeagueWindow.OnInputSourceChanged();
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isToBlock = AddValueInput<bool>("isToBlock");
	}
}
