using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Enable Fighters", 0)]
[Category("Game/Player")]
public class Flow_EnableFighters : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isEnabled;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			if (isEnabled.value)
			{
				Fight.Instance.FightView.FightBehaviour.EnableFighters();
			}
			else
			{
				Fight.Instance.FightView.FightBehaviour.DisableFighters();
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isEnabled = AddValueInput<bool>("isEnabled");
	}
}
