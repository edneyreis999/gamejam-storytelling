using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("UI Vendor Window Item Components\nShow Mask", 0)]
[Category("Game/Tutorial")]
public class Flow_UIVendorWindowItemComponentsShowMask : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isToShow;

	public override string name => "UI Vendor Window Item Components\n" + (isToShow.value ? "Show" : "Hide") + " Mask";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			RectTransform tutorialRectComponents = GUIElements.Get<UIVendorWindow>().TutorialFactoryElement.TutorialRectComponents;
			if (isToShow.value)
			{
				GUIElements.Get<UITutorialWindow>().ShowMask(tutorialRectComponents);
				@out.Call(flow);
			}
			else
			{
				GUIElements.Get<UITutorialWindow>().HideMask(tutorialRectComponents);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isToShow = AddValueInput<bool>("isToShow");
	}
}
