using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Change School Button\nInteractableState\nIn SchoolWidget", 0)]
[Category("Game/UI")]
public class Flow_ChangeSchoolWidgetButtonInteractableState : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isInteractable;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.uiSystem.uiSchoolWidgetData.ChangeSchoolButtonInteractableState(isInteractable.value);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isInteractable = AddValueInput<bool>("isInteractable");
	}
}
