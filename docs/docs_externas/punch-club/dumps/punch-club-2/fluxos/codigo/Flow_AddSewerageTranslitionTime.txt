using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Add Sewerage Translition Time", 0)]
[Category("Game")]
public class Flow_AddSewerageTranslitionTime : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			TimeManager.AddTimeWithAnimation(ConstsDefinition.Get("transition_between_sewerage_rooms").IntValue);
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
	}
}
