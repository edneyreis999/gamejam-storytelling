using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvasNodes;

[Name("Set GPP Bottom Border", 0)]
[Category("Game")]
public class Flow_SetGppBottomBorder : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<float> borderValue;

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.playerData.bottomGppBorder = borderValue.value;
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		borderValue = AddValueInput<float>("borderValue".CapitalizeFirst());
	}
}
