using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("UI League Fighters\nShow Mask", 0)]
[Category("Game/Tutorial")]
public class Flow_UILeagueFighterShowMask : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isToShow;

	public override string name => "UI League Fighters\n" + (isToShow.value ? "Show" : "Hide") + " Mask";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			RectTransform fighterTableRectTransform = GUIElements.Get<UILeagueWindow>().FighterTableRectTransform;
			if (isToShow.value)
			{
				GUIElements.Get<UITutorialWindow>().ShowMask(fighterTableRectTransform);
			}
			else
			{
				GUIElements.Get<UITutorialWindow>().HideMask(fighterTableRectTransform);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isToShow = AddValueInput<bool>("isToShow");
	}
}
