using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;
using UnityEngine;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes.Tutorial;

[Name("UI Prepare To Fight Window\nPerks Show Mask", 0)]
[Category("Game/Tutorial")]
public class Flow_UIPrepareToFightWindowPerksShowMask : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isToShow;

	public override string name => "UI Prepare To Fight Window\nPerks " + (isToShow.value ? "Show" : "Hide") + " Mask";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			RectTransform perkTutorialRect = GUIElements.Get<UIPrepareFightWindow>().PerkTutorialRect;
			if (isToShow.value)
			{
				GUIElements.Get<UITutorialWindow>().ShowMask(perkTutorialRect);
			}
			else
			{
				GUIElements.Get<UITutorialWindow>().HideMask(perkTutorialRect);
			}
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isToShow = AddValueInput<bool>("isToShow");
	}
}
