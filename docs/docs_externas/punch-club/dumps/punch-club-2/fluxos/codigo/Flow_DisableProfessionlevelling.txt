using FlowCanvas;
using ParadoxNotion;
using ParadoxNotion.Design;

namespace PunchClub2.FlowCanvas.FlowCanvasNodes;

[Name("Disable Profession levelling", 0)]
[Category("Game/Player")]
public class Flow_DisableProfessionlevelling : PC2CustomFlowNode
{
	private FlowInput @in;

	private FlowOutput @out;

	private ValueInput<bool> isEnable;

	public override string name => (isEnable.value ? "<color=#8aff92>Enable</color>" : "<color=#ff8a8a>Disable</color>") + " Profession levelling";

	protected override void RegisterPorts()
	{
		@in = AddFlowInput("in".CapitalizeFirst(), delegate(Flow flow)
		{
			MainGame.me.gameSave.professionSystem.isLevelingAvailable = isEnable.value;
			@out.Call(flow);
		});
		@out = AddFlowOutput("out".CapitalizeFirst());
		isEnable = AddValueInput<bool>("isEnable");
	}
}
