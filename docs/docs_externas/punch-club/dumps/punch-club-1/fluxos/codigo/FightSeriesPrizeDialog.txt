using System.Collections.Generic;
using UnityEngine;

public class FightSeriesPrizeDialog : SimpleDialog
{
	public UIProgressBar progress_bar;

	public UIGrid prizes_grid;

	private static GameRes _rew = new GameRes();

	private List<FightSeriesPrizeMark> prizes = new List<FightSeriesPrizeMark>();

	public void Open(CommonStatic.VoidDelegate on_continue, CommonStatic.VoidDelegate on_cancel, string custom_text = "", string custom_prizes = "")
	{
		string[] array = custom_prizes.Split(';');
		if (VHSConst.GAMEPAD_IS_CURRENT_CONTROLLER)
		{
			btn_1.gameObject.SetActive(value: true);
			btn_2.gameObject.SetActive(value: true);
		}
		Open(GJL.L("HDR_SERIES_FIGHT"), GJL.L((!(custom_text == string.Empty)) ? custom_text : "TXT_SERIES_FIGHT"), new ButtonDescription
		{
			color = SimpleButton.ButtonColor.Green,
			text = "(Action1)" + GJL.L("SERIES_CONTINUE"),
			on_click = delegate
			{
				GJTimer.AddTimer(0.14f, delegate
				{
					on_continue();
					MainGame.me.gui_elements.RedrawHUD();
				});
			}
		}, new ButtonDescription
		{
			color = SimpleButton.ButtonColor.Red,
			text = "(Action2)" + GJL.L("SERIES_STOP"),
			on_click = delegate
			{
				on_cancel();
				MainGame.me.gui_elements.RedrawHUD();
			}
		});
		if (FightScene.fight_series_step == 1)
		{
			btn_2.SetEnabled(is_enabled: false);
			if (VHSConst.GAMEPAD_IS_CURRENT_CONTROLLER)
			{
				btn_2.gameObject.SetActive(value: false);
			}
		}
		if (FightScene.fight_series_step >= 6)
		{
			btn_1.SetEnabled(is_enabled: false);
			if (VHSConst.GAMEPAD_IS_CURRENT_CONTROLLER)
			{
				btn_1.gameObject.SetActive(value: false);
			}
		}
		progress_bar.value = 0.146f + (float)(FightScene.fight_series_step - 1) * 0.171f;
		prizes.Clear();
		prizes.AddRange(prizes_grid.gameObject.GetComponentsInChildren<FightSeriesPrizeMark>(includeInactive: true));
		_rew = null;
		int num = -1;
		int num2;
		for (num2 = 0; num2 < 6; num2++)
		{
			GameObject gameObject = prizes[num2].ok_sign.gameObject;
			if (num2 % 2 == 0)
			{
				gameObject.SetActive(value: false);
				continue;
			}
			bool flag = num2 < FightScene.fight_series_step;
			gameObject.SetActive(flag);
			if (flag)
			{
				num = num2;
			}
		}
		num2 = -1;
		foreach (int item in PrefightGUI._cur_fight_data.rew_n)
		{
			num2++;
			if (num2 >= PrefightGUI._cur_fight_data.rew_r.Count)
			{
				num2 = PrefightGUI._cur_fight_data.rew_r.Count - 1;
			}
			string text = ((num2 >= array.Length) ? string.Empty : array[num2]);
			string text2 = ManagerGUI.PrepareFamePrice(PrefightGUI._cur_fight_data.rew_r[num2].ToDescStrOutlined("\n"));
			if (text.Length > 0)
			{
				text2 = text2 + "\n" + text;
			}
			prizes[item - 1].prize.text = text2;
			if (item - 1 == num)
			{
				_rew = PrefightGUI._cur_fight_data.rew_r[num2];
			}
		}
		if (_rew == null)
		{
			_rew = new GameRes();
		}
	}

	public static void GetMyPrize(bool lost_game = false)
	{
		if (lost_game)
		{
			_rew = new GameRes();
		}
		MainGame.me.gui_elements.end_of_fight.ResetEnduranceMode();
		Sound.ContinueTrainMusic();
		string txt = GJL.L("TXT_SERIES_LOST");
		if (!lost_game)
		{
			txt = GJL.L("TXT_SERIES_END");
		}
		GJCommons.VoidDelegate on_close = delegate
		{
			Debug.Log("on_close");
			MainGame.save.ch.AddCharRes(_rew);
			MainGame.me.quest_system.CheckKeyQuests("series_end_" + PrefightGUI.last_fight_id);
			MainGame.me.gui_elements.RedrawHUD();
		};
		string pic = ((ManagerGUI.GetManagerType() != UpgradeData.ManagerType.Star) ? "pic_map" : "pic_cinema");
		if (PrefightGUI.last_fight_id == 1005 || PrefightGUI.last_fight_id == 1008)
		{
			pic = "pic_base";
		}
		MainGame.me.gui_elements.pic_dialog.OpenPrizeDialog(GJL.L("HDR_SERIES_FIGHT"), txt, pic, _rew, delegate
		{
			on_close();
		});
	}
}
