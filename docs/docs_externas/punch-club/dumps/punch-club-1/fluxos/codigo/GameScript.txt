	public static bool IsAnyScriptRunning()
	{
		return _total_executed_scripts > 0;
	}

	public static void CancelAllScripts()
	{
		if (!IsAnyScriptRunning())
		{
			return;
		}
		Debug.Log("Cancelling scripts, running = " + _total_executed_scripts);
		int num = 0;
		do
		{
			CustomUScript customUScript = UnityEngine.Object.FindObjectOfType<CustomUScript>();
			if (customUScript == null)
			{
				break;
			}
			customUScript.TerminateMe();
		}
		while (num++ <= 30);
		_total_executed_scripts = 0;
		_all_scripts.Clear();
	}

		_lua = null;

	public void FromLuaExpression(string expression)
	{
		FromLua("return " + expression + "\n ");
		_is_expression = true;
	}

	public float ExecuteExpression()
	{
		if (!_is_expression)
		{
			Debug.LogError("Trying execute non-expression Lua script as an expression");
			return 0f;
		}
		if (_lua.L_DoString(_lua_code) != ThreadStatus.LUA_OK)
		{
			Debug.LogError("Error executing Lua expression : " + _lua.ToString());
			Debug.LogWarning("LUA code: " + _lua_code);
			throw new Exception("Error executing Lua expression");
		}
		float result = (float)_lua.ToNumber(-1);
		_lua.Pop(1);
		return result;
	}

	public void Execute()
	{
		if (_is_expression)
		{
			Debug.LogError("Trying execute expression Lua script as non-expression");
			return;
		}
		id = UnityEngine.Random.Range(int.MinValue, int.MaxValue);
		if (_uscript_name != string.Empty)
		{
			GameScriptStatic.OnuScriptStarted(_uscript_name);
			uscript = CustomUScript.Create(_uscript_name);
			TwitchGameEngine.OnGlobalEvent(TwitchGameEngine.GlobalEvent.ScriptStarted);
			return;
		}
		if (_lua == null)
		{
			Debug.Log("<color=yellow>lua script is null, exiting...</color>");
			return;
		}
		Debug.Log("<color=yellow>executing script... </color>" + id + ", executed scripts: " + _total_executed_scripts + ", expression: " + _is_expression);
		if (!_is_expression)
		{
			_total_executed_scripts++;
			_all_scripts.Add(this);
		}
		_thread = _lua.NewThread();
		_status = _thread.L_LoadString(_lua_code);
		if (_status != ThreadStatus.LUA_OK)
		{
			Debug.LogError("Error parsing lua code: " + _status);
			Debug.LogWarning("LUA code:\n" + _lua_code);
			return;
		}
		try
		{
			_thread.Resume(null, 0);
		}
		catch (LuaRuntimeException ex)
		{
			Debug.LogError(string.Concat("Error executing lua code #", ex.ErrCode, "\n", ex.ToString()));
		}
	}

	private int DoContinue(ILuaState lua)
	{
		int num = lua.GetTop();
		if (num > 0)
		{
			_thread.Resume(null, 0);
			num--;
		}
		return num;
	}

	public int ReturnLuaYield(ILuaState s)
	{
		_temp_state = s;
		return s.YieldK(s.GetTop(), 0, null);
	}

	public void DoContinueDelegate()
	{
		DoContinue(_temp_state);
	}

	private int OnScriptFinished(ILuaState state)
	{
		if (!_is_expression)
		{
			_total_executed_scripts--;
			Debug.Log("<color=yellow>finished script </color>" + id + ", executed scripts: " + _total_executed_scripts + ", expression: " + _is_expression);
			if (_all_scripts.Contains(this))
			{
				_all_scripts.Remove(this);
			}
		}
		return 1;
	}

	public void FromString(string script, string on_end_function = null)
	{
		if (IsUScript(script))
		{
			FromUScriptName(script.Substring(1));
		}
		else
		{
			FromLua(script + ((on_end_function != null) ? ("\n vhs." + on_end_function + "()") : string.Empty));
		}
	}

	private static bool IsUScript(string str)
	{
		return str.Length > 1 && str[0] == ':';
	}
