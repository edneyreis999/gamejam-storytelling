using System;
using DG.Tweening;
using DG.Tweening.Core;
using UnityEngine;

public class SpeechBubbleInsideGfx : MonoBehaviour
{
	public UIWidget corner_speak;

	public UIWidget corner_think;

	public UIWidget background;

	public MyTooltipContents contents;

	private bool _mirror_x;

	private bool _mirror_y;

	private bool _animating = true;

	private UILabel _lbl_text;

	private UITable _table;

	private string _txt;

	private int _cur_word;

	private float _last_time;

	private int _words_drawn = -1;

	private int _max_height;

	private bool _started_text;

	private const int MAX_WIDTH = 100;

	private const float BUBBLE_APPEAR_TIME = 0.32f;

	public bool deleted;

	public void SetType(SpeechBubbleGUI.Type type)
	{
		corner_speak.gameObject.SetActive(type == SpeechBubbleGUI.Type.Speak);
		corner_think.gameObject.SetActive(type == SpeechBubbleGUI.Type.Think);
		switch (type)
		{
		case SpeechBubbleGUI.Type.Speak:
			base.transform.localPosition = new Vector3(0f, 0f, 0f);
			break;
		case SpeechBubbleGUI.Type.None:
			base.transform.localPosition = new Vector3(0f, -10f, 0f);
			break;
		case SpeechBubbleGUI.Type.Think:
			base.transform.localPosition = new Vector3(12f, 16f, 0f);
			break;
		}
	}

	public void SetText(string txt, bool animated = true)
	{
		_txt = txt;
		_animating = animated;
		contents.SetContents(new MyTooltipContents.Content[1]
		{
			new MyTooltipContents.Content
			{
				txt = txt,
				type = MyTooltipContents.ContentType.Text
			}
		});
		_lbl_text = GetComponentInChildren<UILabel>();
		_table = GetComponentInChildren<UITable>();
		bool mirror_x = false;
		bool mirror_y = false;
		Vector2 contentsSize = contents.GetContentsSize();
		Vector2 vector = base.transform.parent.transform.localPosition;
		if (vector.y + contentsSize.y + 15f > 180f)
		{
			mirror_y = true;
		}
		if (vector.x + contentsSize.x + 7f > 320f)
		{
			mirror_x = true;
		}
		SetMirror(mirror_x, mirror_y);
		GetComponentInChildren<MyTooltipContents>().is_speech_text = animated;
		GetComponentInChildren<MyTooltipContents>().min_x = 0;
		int min_x = Mathf.RoundToInt(contents.GetContentsSize().x);
		if (min_x != _lbl_text.width)
		{
			_lbl_text.width = min_x;
			contentsSize = contents.GetContentsSize();
			contents.container.height = (int)contentsSize.y;
		}
		Init();
		UIWidget wgt = _table.transform.parent.GetComponent<UIWidget>();
		_max_height = wgt.height;
		if (!_animating)
		{
			return;
		}
		DoUpdate(animate_size_change: false);
		wgt.height = 5;
		_lbl_text.text = string.Empty;
		contents.GetContentsSize();
		int width = contents.container.width;
		contents.container.width = 10;
		DOTween.To(() => contents.container.width, delegate(int a)
		{
			contents.container.width = a;
		}, min_x, 0.32f).SetEase(Ease.OutBack);
		Tweener tweener = DOTween.To(() => wgt.height, delegate(int a)
		{
			wgt.height = a;
		}, _max_height, 0.32f);
		tweener.SetEase(Ease.OutBack);
		tweener.onComplete = (TweenCallback)Delegate.Combine(tweener.onComplete, (TweenCallback)delegate
		{
			if (!deleted)
			{
				GetComponentInChildren<MyTooltipContents>().min_x = min_x;
				_started_text = true;
				_last_time = GJCommons.GetTicksInSeconds();
			}
		});
	}

	private void Init()
	{
		_last_time = GJCommons.GetTicksInSeconds();
		_cur_word = 0;
		_words_drawn = -1;
		_started_text = false;
	}

	public void Update()
	{
		DoUpdate(animate_size_change: true);
	}

	public void DoUpdate(bool animate_size_change)
	{
		if (!_animating || _lbl_text == null || !_started_text)
		{
			return;
		}
		int num = Mathf.FloorToInt((GJCommons.GetTicksInSeconds() - _last_time) / 0.02f);
		if (num > _txt.Length)
		{
			num = _txt.Length;
		}
		if (num != _words_drawn)
		{
			_words_drawn = num;
			if (_words_drawn < 0)
			{
				_words_drawn = 0;
			}
			_lbl_text.text = _txt.Substring(0, _words_drawn) + "[c][00000000]" + _txt.Substring(_words_drawn);
			UIWidget component = _table.transform.parent.GetComponent<UIWidget>();
			int height = component.height;
			_table.Reposition();
			component.height = _max_height;
		}
	}

	public void ForceMirror(bool force_mirror_x, bool force_mirror_y)
	{
		SetMirror(_mirror_x || force_mirror_x, _mirror_y || force_mirror_y);
	}

	private void SetMirror(bool mirror_x, bool mirror_y)
	{
		_mirror_x = mirror_x;
		_mirror_y = mirror_y;
		int num = ((!mirror_x) ? 1 : (-1));
		int num2 = ((!mirror_y) ? 1 : (-1));
		Vector2 contentsSize = contents.GetContentsSize();
		base.transform.parent.gameObject.transform.localScale = new Vector3(num, num2, 1f);
		contents.container.GetComponentInChildren<UITable>().transform.localScale = new Vector3(num, num2, 1f);
		contents.container.GetComponentInChildren<UITable>().transform.localPosition = new Vector3((!mirror_x) ? 0f : contentsSize.x, (!mirror_y) ? 4f : (contentsSize.y - 5f));
	}

	public bool IsRightSideOutsideScreen(SpeechBubbleGUI obj)
	{
		return (MainGame.me.gui_elements.GetComponent<Camera>().WorldToScreenPoint(new Vector3(obj.transform.localPosition.x, 0f, 0f)) + new Vector3(background.width * 2, 0f, 0f)).x >= (float)Screen.width;
	}
}
