public class uScriptAct_GetCurrentActionName : uScriptLogic
{
	public bool Out => true;

	public void In([FriendlyName("Action name", "Current action name")] out string act_name)
	{
		ActionData current_action = MainGame.save.ch.char_gfx.current_action;
		act_name = ((current_action != null) ? current_action.name : string.Empty);
	}
}
