using System;
using UnityEngine;

[NodePath("Actions/VHS Story/")]
[NodeCopyright("Copyright 2015 by GameJam")]
[NodeToolTip("")]
[NodeAuthor("GameJam", "http://www.gamejam.ru")]
[FriendlyName("Anim control", "")]
public class uScriptAct_AnimationController : uScriptLogic
{
	public enum SetWhat
	{
		Trigger,
		Int
	}

	public bool Out => true;

	[FriendlyName("Event")]
	public event uScript_GameObject.uScriptEventHandler Event;

	public void In([FriendlyName("GameObj", "GameObject name")] string go_name, [FriendlyName("what", "")][DefaultValue(SetWhat.Trigger)] SetWhat what, [FriendlyName("Param", "Param name")] string param_name, [FriendlyName("Value", "")][SocketState(false, false)] int value)
	{
		Debug.Log("Animation control: " + go_name + " , set " + param_name);
		GameObject gameObject = GameObject.Find(go_name);
		if (gameObject == null)
		{
			Debug.LogError("Game object '" + go_name + "' not found.");
			return;
		}
		AnimationBackEvent animationBackEvent = gameObject.GetComponent<AnimationBackEvent>();
		if (animationBackEvent == null)
		{
			animationBackEvent = gameObject.AddComponent<AnimationBackEvent>();
		}
		animationBackEvent.animation_event = delegate
		{
			if (this.Event != null)
			{
				this.Event(this, new EventArgs());
			}
		};
		Animator component = gameObject.GetComponent<Animator>();
		if (component == null)
		{
			Debug.LogError("Animator for object " + go_name + " not found");
			return;
		}
		switch (what)
		{
		case SetWhat.Trigger:
			component.SetTrigger(param_name);
			break;
		case SetWhat.Int:
			component.SetInteger(param_name, value);
			break;
		}
	}
}
